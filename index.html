webhook-engine / src / utils / hmac.js
const crypto = require('crypto');

function signPayload(payload, secret) {
  return crypto
    .createHmac('sha256', secret)
    .update(payload)
    .digest('hex');
}

function verifySignature(payload, signature, secret) {
  const expected = signPayload(payload, secret);
  try {
    // Explicit protection against timing attacks for cryptographic comparison
    return crypto.timingSafeEqual(
      Buffer.from(expected, 'hex'),
      Buffer.from(signature, 'hex')
    );
  } catch {
    return false;
  }
}
webhook-engine / design-sample / worker.go
package main

import (
	"log"
	"net/http"
)

// Job queue pattern for fail-safe background webhook delivery
var eventQueue = make(chan Event, 1000)

func init() {
	// Spin up a highly concurrent worker pool (Goroutines) 
	// to prevent memory leaks from unhandled requests
	for i := 0; i < 5; i++ {
		go deliveryWorker(eventQueue)
	}
}

func handleWebhook(w http.ResponseWriter, r *http.Request) {
	event, err := parsePayload(r)
	if err != nil {
		http.Error(w, "Invalid payload format", http.StatusBadRequest)
		return
	}

	// Non-blocking channel drop to return 202 instantly to the client
	select {
	case eventQueue <- event:
		w.WriteHeader(http.StatusAccepted)
	default:
		// Fails fast if worker pool is completely saturated
		http.Error(w, "Queue capacity exceeded", http.StatusServiceUnavailable)
	}
}
